  Future<void> _loadAccounts() async {

      setState(() {
        loadingAccounts = true;
      });

  Future<void> _loadSchoolYear() async {
    try {
      final String savedSchoolYear = await archiveService
          .getCurrentSchoolYear();

      final List<String> archivedYears = await archiveService
          .getArchivedSchoolYears();

      final List<String> choices = _createSchoolYearChoices();

      if (_isValidSchoolYear(savedSchoolYear)) {
        choices.add(savedSchoolYear);
      }

      choices.removeWhere((String year) => archivedYears.contains(year));

      final List<String> uniqueChoices = choices.toSet().toList()..sort();

      if (!mounted) return;

      final bool savedYearAvailable =
          _isValidSchoolYear(savedSchoolYear) &&
          !archivedYears.contains(savedSchoolYear);

      setState(() {
        archivedSchoolYears = archivedYears;
        availableSchoolYears = uniqueChoices;

        if (savedYearAvailable) {
          currentSchoolYear = savedSchoolYear;
          selectedSchoolYear = savedSchoolYear;
        } else {
          currentSchoolYear = 'Not Selected';
          selectedSchoolYear = null;
        }

        loadingSchoolYear = false;
      });
    } catch (error) {
      if (!mounted) return;

      setState(() {
        currentSchoolYear = 'Unavailable';
        selectedSchoolYear = null;
        loadingSchoolYear = false;
      });
    }
  }

  bool _isValidSchoolYear(String value) {
    final RegExpMatch? match = RegExp(r'^(\d{4})-(\d{4})$').firstMatch(value);

    if (match == null) {
      return false;
    }

    final int? startYear = int.tryParse(match.group(1) ?? '');
    final int? endYear = int.tryParse(match.group(2) ?? '');

    if (startYear == null || endYear == null) {
      return false;
    }

    return endYear == startYear + 1;
  }

  List<String> _createSchoolYearChoices() {
    final int currentYear = DateTime.now().year;

    return List<String>.generate(11, (int index) {
      final int startYear = currentYear - 3 + index;

      return '$startYear-${startYear + 1}';
    });
  }

  Future<void> _setActiveSchoolYear() async {
    final String? schoolYear = selectedSchoolYear;

    if (schoolYear == null) {
      return;
    }

    if (archivedSchoolYears.contains(schoolYear)) {
      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          const SnackBar(
            content: Text('Archived school years cannot be used again.'),
          ),
        );

      return;
    }

    try {
      await archiveService.setCurrentSchoolYear(schoolYear);

      if (!mounted) return;

      setState(() {
        currentSchoolYear = schoolYear;
      });

      await _refreshPerformanceSummary();
      await _refreshRecentActivities();

      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(content: Text('School Year $schoolYear is now active.')),
        );
    } catch (error) {
      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(error.toString().replaceFirst('Exception: ', '')),
          ),
        );
    }
  }

    final String? result = await showDialog<String>(
      context: context,

        return AlertDialog(
          backgroundColor: Colors.white,
          shape: RoundedRectangleBorder(
            borderRadius: BorderRadius.circular(20),
          ),
          title: const Text(
            'Add School Year',
            style: TextStyle(
              color: dashboardBrown,
              fontWeight: FontWeight.w700,
            ),
          ),
          content: TextField(
            controller: controller,
            keyboardType: TextInputType.number,
            maxLength: 4,
            autofocus: true,
            decoration: InputDecoration(
              labelText: 'Starting Year',
              hintText: 'Example: 2027',
              counterText: '',
              filled: true,
              fillColor: const Color(0xFFFBF7F2),
              border: OutlineInputBorder(
                borderRadius: BorderRadius.circular(12),
              ),
            ),
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.pop(dialogContext);
              },
              child: const Text('Cancel'),

            ElevatedButton(
              onPressed: () {
                Navigator.pop(dialogContext, controller.text.trim());
              },

                foregroundColor: Colors.white,
              ),
              child: const Text('Add'),
            ),
          ],
        );
      },
    );

    controller.dispose();

    if (result == null) return;

    final int? startYear = int.tryParse(result);

    if (startYear == null || result.length != 4) {
      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          const SnackBar(content: Text('Enter a valid 4-digit starting year.')),
        );

      return;
    }

    final String schoolYear = '$startYear-${startYear + 1}';

    if (archivedSchoolYears.contains(schoolYear)) {
      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(content: Text('$schoolYear is already archived.')),
        );

      return;
    }

    if (!mounted) return;

    setState(() {
      if (!availableSchoolYears.contains(schoolYear)) {
        availableSchoolYears.add(schoolYear);
        availableSchoolYears.sort();
      }

      selectedSchoolYear = schoolYear;
    });

    ScaffoldMessenger.of(context)
      ..hideCurrentSnackBar()
      ..showSnackBar(
        SnackBar(
          content: Text(
            '$schoolYear added. Press Set School Year to activate it.',
          ),

      );

  void _resetSchoolYearPerformanceData() {

    recentActivities = [];

  // ============================================================

  // ============================================================

  Future<void> _refreshPerformanceSummary() async {
    if (!_isValidSchoolYear(currentSchoolYear)) {
      if (!mounted) return;
      setState(() {
        _resetSchoolYearPerformanceData();
      });
      return;
    }

    try {
      final PerformanceService? service = performanceService;

      if (service == null) {
        if (!mounted) return;
        setState(() {
          _resetSchoolYearPerformanceData();
        });
        return;
      }

      final List<StudentPerformanceModel> performances = await service
          .getPerformances(schoolYear: currentSchoolYear);

      if (!mounted) return;

      if (performances.isEmpty) {
        setState(() {
          _resetSchoolYearPerformanceData();
        });
        return;
      }

      double englishTotal = 0;
      double mathematicsTotal = 0;
      double scienceTotal = 0;
      double overallTotal = 0;

      int excellent = 0;
      int good = 0;
      int average = 0;
      int needsImprovement = 0;

      for (final StudentPerformanceModel performance in performances) {
        englishTotal += performance.englishAverage;
        mathematicsTotal += performance.mathematicsAverage;
        scienceTotal += performance.scienceAverage;
        overallTotal += performance.overallAverage;

        final double score = performance.overallAverage;

        if (score >= 90) {
          excellent++;
        } else if (score >= 75) {
          good++;
        } else if (score >= 60) {
          average++;
        } else {
          needsImprovement++;
        }
      }

      final double count = performances.length.toDouble();

      setState(() {
        averageScore = (overallTotal / count).round();

        learningAreaScores = [
          englishTotal / count,
          mathematicsTotal / count,
          scienceTotal / count,
        ];

        scoreCategories = [
          _ScoreCategory(
            label: 'Excellent',
            count: excellent,
            color: const Color(0xFF3CA667),
          ),
          _ScoreCategory(
            label: 'Good',
            count: good,
            color: const Color(0xFF6DA9DF),
          ),
          _ScoreCategory(
            label: 'Average',
            count: average,
            color: const Color(0xFFE4A849),
          ),
          _ScoreCategory(
            label: 'Needs Improvement',
            count: needsImprovement,
            color: const Color(0xFFD87567),
          ),
        ];
      });
    } catch (error) {
      if (!mounted) return;

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(content: Text('Unable to load performance data: ')),
        );
    }
  }

  Future<void> _archiveCurrentSchoolYear() async {
    if (archivingSchoolYear) return;

    setState(() {
      archivingSchoolYear = true;
    });

    final String yearToArchive = currentSchoolYear;

    try {
      final DateTime archivedAt = DateTime.now();

      final List<ArchivedStudentSummary> summaries = [];

      for (final StudentModel student in students) {
        final StudentPerformanceModel? performance = performanceService == null
            ? null
            : await performanceService!.getStudentPerformance(
                studentId: student.id,
                schoolYear: yearToArchive,
              );

        summaries.add(
          ArchivedStudentSummary(
            studentId: student.id,
            studentName: student.fullName,
            schoolYear: yearToArchive,
            grade: student.grade,
            condition: student.condition,
            englishAverage: performance?.englishAverage ?? 0,
            mathematicsAverage: performance?.mathematicsAverage ?? 0,
            scienceAverage: performance?.scienceAverage ?? 0,
            overallAverage: performance?.overallAverage ?? 0,
            activitiesCompleted: performance?.activitiesCompleted ?? 0,
            totalActivities: performance?.totalActivities ?? 0,
            totalAttempts: performance?.totalAttempts ?? 0,
            completionStatus: performance?.completionStatus ?? 'Not Started',
            archivedAt: archivedAt,
          ),
        );
      }
      await archiveService.archiveSchoolYear(
        schoolYear: yearToArchive,
        students: summaries,
      );

      if (performanceService != null) {
        await performanceService!.clearSchoolYearPerformance(yearToArchive);
      }

      final ActivityService? activityService = widget.activityService;

      if (activityService != null) {
        await activityService.clearSchoolYearActivities(yearToArchive);
      }

      final List<String> updatedArchivedYears = await archiveService
          .getArchivedSchoolYears();

      if (!mounted) return;

      setState(() {
        archivedSchoolYears = updatedArchivedYears;

        availableSchoolYears.removeWhere(
          (String year) => updatedArchivedYears.contains(year),
        );

        currentSchoolYear = 'Not Selected';
        selectedSchoolYear = null;

        _resetSchoolYearPerformanceData();

        archivingSchoolYear = false;
      });

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(
              'School Year $yearToArchive archived successfully. '
              'Choose the next active school year.',
            ),
          ),
        );
    } catch (error) {
      if (!mounted) return;

      setState(() {
        archivingSchoolYear = false;
      });

      final String message = error.toString().replaceFirst('Exception: ', '');

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(SnackBar(content: Text(message)));
    }
  }

  // ============================================================
  // LOAD RECENT ACTIVITY
  // ============================================================

  Future<void> _refreshRecentActivities() async {
    final ActivityService? service = widget.activityService;

    if (service == null || !_isValidSchoolYear(currentSchoolYear)) {
      if (!mounted) return;
      setState(() {
        recentActivities = <_ActivityData>[];
      });
      return;
    }

    try {
      final records = await service.getRecentActivities(
        schoolYear: currentSchoolYear,
        limit: 10,
      );

      if (!mounted) return;

      setState(() {
        recentActivities = records.map((record) {
          return _ActivityData(
            time: _formatActivityTime(record.completedAt),
            student: record.studentName,
            activity: record.activityName,
            learningArea: record.learningArea,
            score: '${record.score.round()}%',
          );
        }).toList();
      });
    } catch (error) {
      if (!mounted) return;
      setState(() {
        recentActivities = <_ActivityData>[];
      });
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            'Unable to load recent activity: ${error.toString().replaceFirst('Exception: ', '')}',
          ),
        ),
      );
    }
  }

  String _formatActivityTime(DateTime value) {
    final DateTime time = value.toLocal();
    final DateTime now = DateTime.now();

    final bool today =
        time.year == now.year && time.month == now.month && time.day == now.day;

    final DateTime yesterday = now.subtract(const Duration(days: 1));

    final bool wasYesterday =
        time.year == yesterday.year &&
        time.month == yesterday.month &&
        time.day == yesterday.day;

    if (wasYesterday) {
      return 'Yesterday';
    }

    if (!today) {
      return '${time.month}/${time.day}/${time.year}';
    }

    final int hour = time.hour == 0
        ? 12
        : time.hour > 12
        ? time.hour - 12
        : time.hour;
    final String minute = time.minute.toString().padLeft(2, '0');
    final String period = time.hour >= 12 ? 'PM' : 'AM';

    return '$hour:$minute $period';
  }
