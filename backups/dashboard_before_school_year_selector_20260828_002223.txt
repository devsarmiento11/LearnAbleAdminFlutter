  Future<void> _loadAccounts() async {
    if (mounted) {
      setState(() {
        loadingAccounts = true;
      });

      final List<StudentModel> loadedStudents = await accountService
          .getStudents();

        loadingAccounts = false;

        SnackBar(
          content: Text(error.toString()),

  Future<void> _loadSchoolYear() async {

  // ============================================================
  // ARCHIVE SCHOOL YEAR CONFIRMATION
  // ============================================================

  // ============================================================
  // NEXT SCHOOL YEAR
  // ============================================================

  String _getNextSchoolYear(String schoolYear) {
    final List<String> parts = schoolYear.split('-');

    if (parts.length != 2) {
      throw Exception('Invalid school year format.');
    }

    final int? startYear = int.tryParse(parts[0]);

    final int? endYear = int.tryParse(parts[1]);

    if (startYear == null || endYear == null) {
      throw Exception('Invalid school year format.');
    }

    return '${startYear + 1}-${endYear + 1}';
  }

  // ============================================================
  // ARCHIVE CURRENT SCHOOL YEAR
  // ============================================================

  Future<void> _archiveCurrentSchoolYear() async {
    if (archivingSchoolYear) return;

    setState(() {
      archivingSchoolYear = true;
    });

    final String yearToArchive = currentSchoolYear;

    try {
      final DateTime archivedAt = DateTime.now();

      final List<ArchivedStudentSummary> summaries = students.map((
        StudentModel student,
      ) {
        return ArchivedStudentSummary(
          studentId: student.id,
          studentName: student.fullName,
          schoolYear: yearToArchive,
          grade: student.grade,
          condition: student.condition,

          // Real Firebase/Unity scores will replace
          // these temporary zero values later.
          englishAverage: 0,
          mathematicsAverage: 0,
          scienceAverage: 0,
          overallAverage: 0,

          activitiesCompleted: 0,
          totalActivities: 0,
          totalAttempts: 0,

          completionStatus: 'Not Started',

          archivedAt: archivedAt,
        );
      }).toList();

      await archiveService.archiveSchoolYear(
        schoolYear: yearToArchive,
        students: summaries,
      );

      final String nextSchoolYear = _getNextSchoolYear(yearToArchive);

      await archiveService.setCurrentSchoolYear(nextSchoolYear);

      if (!mounted) return;

      setState(() {
        currentSchoolYear = nextSchoolYear;

        archivingSchoolYear = false;
      });

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(
          SnackBar(
            content: Text(
              'School Year $yearToArchive archived successfully. '
              'Current School Year is now $nextSchoolYear.',
            ),
          ),
        );
    } catch (error) {
      if (!mounted) return;

      setState(() {
        archivingSchoolYear = false;
      });

      final String message = error.toString().replaceFirst('Exception: ', '');

      ScaffoldMessenger.of(context)
        ..hideCurrentSnackBar()
        ..showSnackBar(SnackBar(content: Text(message)));
    }
  }

          alignment: BarChartAlignment.spaceAround,

          barTouchData: BarTouchData(enabled: true),

          borderData: FlBorderData(show: false),

          gridData: FlGridData(
            show: true,
            drawVerticalLine: false,
            horizontalInterval: 20,

            getDrawingHorizontalLine: (double value) {
              return const FlLine(color: dashboardBorder, strokeWidth: 1);
            },
          ),

          titlesData: FlTitlesData(
            topTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),

            rightTitles: const AxisTitles(
              sideTitles: SideTitles(showTitles: false),
            ),

            leftTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 37,
                interval: 20,

                getTitlesWidget: (double value, TitleMeta meta) {
                  return SideTitleWidget(
                    meta: meta,
                    space: 5,

                    child: Text(
                      '${value.toInt()}%',

                      style: const TextStyle(
                        color: dashboardMuted,

                        fontSize: 9,
                      ),
                    ),
                  );
                },
              ),
            ),

            bottomTitles: AxisTitles(
              sideTitles: SideTitles(
                showTitles: true,
                reservedSize: 38,

                getTitlesWidget: (double value, TitleMeta meta) {
                  final int index = value.toInt();

                  if (index < 0 || index >= learningAreaLabels.length) {
                    return const SizedBox.shrink();
                  }

                  String label = learningAreaLabels[index];

                  if (label == 'Mathematics') {
                    label = 'Math';
                  }

                  return SideTitleWidget(
                    meta: meta,
                    space: 8,

                    child: Text(
                      label,

                      style: const TextStyle(
                        color: dashboardMuted,

                        fontSize: 9,

                        fontWeight: FontWeight.w500,
                      ),
                    ),
                  );
                },
              ),
            ),
          ),

          barGroups: List.generate(learningAreaScores.length, (int index) {
            return BarChartGroupData(
              x: index,

              barRods: [
                BarChartRodData(
                  toY: learningAreaScores[index],

                  width: 24,

                  color: dashboardAccent,

                  borderRadius: BorderRadius.circular(7),
                ),
              ],
            );
          }),
